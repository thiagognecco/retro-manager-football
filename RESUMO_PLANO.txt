   ```python
   import cProfile, pstats, io
   from match_simulation_engine_v2 import Match, create_sample_teams

   pr = cProfile.Profile()
   pr.enable()
   m = Match(*create_sample_teams())
   m.simulate_match()
   pr.disable()

   s = io.StringIO()
   ps = pstats.Stats(pr, stream=s).sort_stats('cumulative')
   ps.print_stats(20)
   print(s.getvalue())
   ```
